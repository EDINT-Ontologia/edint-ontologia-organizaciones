ORG01.sparql
ORG01: ¿Qué organizaciones son entidades de derecho público?
PREFIX edintorg: <http://vocab.linkeddata.es/datosabiertos/def/sector-publico/organizacion#>
PREFIX rdfs:     <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?organization ?label
WHERE {
  ?organization a edintorg:PublicLawEntity .

  OPTIONAL {
    ?organization rdfs:label ?label .
    FILTER(LANG(?label) = "es" || LANG(?label) = "")
  }
}
ORDER BY ?label
ORG02.sparql
ORG02: ¿Qué unidades orgánicas dependen de una entidad de derecho público?
PREFIX edintorg: <http://vocab.linkeddata.es/datosabiertos/def/sector-publico/organizacion#>
PREFIX rdfs:     <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?organicUnit ?organicUnitLabel ?publicLawEntity ?publicLawEntityLabel
WHERE {
  ?organicUnit a edintorg:OrganicUnit ;
               edintorg:dependsOnRoot ?publicLawEntity .

  ?publicLawEntity a edintorg:PublicLawEntity .

  OPTIONAL {
    ?organicUnit rdfs:label ?organicUnitLabel .
    FILTER(LANG(?organicUnitLabel) = "es" || LANG(?organicUnitLabel) = "")
  }

  OPTIONAL {
    ?publicLawEntity rdfs:label ?publicLawEntityLabel .
    FILTER(LANG(?publicLawEntityLabel) = "es" || LANG(?publicLawEntityLabel) = "")
  }
}
ORDER BY ?publicLawEntityLabel ?organicUnitLabel

ORG09.sparql
ORG09: ¿Qué organizaciones están dadas de baja o han dejado de estar activas?
PREFIX edintorg: <http://vocab.linkeddata.es/datosabiertos/def/sector-publico/organizacion#>
PREFIX org:      <http://www.w3.org/ns/org#>
PREFIX skos:     <http://www.w3.org/2004/02/skos/core#>
PREFIX rdfs:     <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?organization ?organizationLabel ?status ?statusLabel ?deregistrationNote
WHERE {
  ?organization a org:Organization ;
                edintorg:hasOrganizationStatus ?status .

  ?status skos:prefLabel ?statusLabel .

  FILTER(LANG(?statusLabel) = "es" || LANG(?statusLabel) = "")
  FILTER(
    CONTAINS(LCASE(STR(?statusLabel)), "Anulado") ||
    CONTAINS(LCASE(STR(?statusLabel)), "Extinguido")
  )

  OPTIONAL {
    ?organization edintorg:deregistrationNote ?deregistrationNote .
  }

  OPTIONAL {
    ?organization rdfs:label ?organizationLabel .
    FILTER(LANG(?organizationLabel) = "es" || LANG(?organizationLabel) = "")
  }
}
ORDER BY ?organizationLabel

ORG10: ¿Qué personas ocupan un rol dentro de una organización?
PREFIX org:  <http://www.w3.org/ns/org#>
PREFIX foaf: <http://xmlns.com/foaf/0.1/>
PREFIX skos: <http://www.w3.org/2004/02/skos/core#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?person ?personName ?organization ?organizationLabel ?role ?roleLabel
WHERE {

  # Sustituir por la organización concreta si se quiere limitar a una organización:
  # VALUES ?organization { <URI_DE_LA_ORGANIZACION> }

  ?membership a org:Membership ;
              org:member ?person ;
              org:organization ?organization ;
              org:role ?role .

  ?person a foaf:Person .

  OPTIONAL {
    ?person foaf:name ?personName .
  }

  OPTIONAL {
    ?organization rdfs:label ?organizationLabel .
    FILTER(LANG(?organizationLabel) = "es" || LANG(?organizationLabel) = "")
  }

  OPTIONAL {
    ?role skos:prefLabel ?roleLabel .
    FILTER(LANG(?roleLabel) = "es" || LANG(?roleLabel) = "")
  }

}
ORDER BY ?organizationLabel ?roleLabel ?personName
